cmake_minimum_required(VERSION 3.20)
project(SteamFrameAV1Lab LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SteamFrameAV1Backend SHARED av1_backend.cpp legacy_picture.cpp legacy_api_bridge.cpp steamvr_trial.cpp
 ../vendor/minhook/src/hook.c ../vendor/minhook/src/buffer.c ../vendor/minhook/src/trampoline.c ../vendor/minhook/src/hde/hde64.c)
target_include_directories(SteamFrameAV1Backend PRIVATE ../vendor ../vendor/minhook/include)
target_link_libraries(SteamFrameAV1Backend PRIVATE d3d11)
target_compile_options(SteamFrameAV1Backend PRIVATE /W4 /permissive- /EHsc)
add_executable(av1_backend_test backend_test.cpp)
target_link_libraries(av1_backend_test PRIVATE SteamFrameAV1Backend d3d11 d3dcompiler winmm ws2_32 avrt)
add_executable(av1_legacy_picture_test legacy_picture_test.cpp)
target_include_directories(av1_legacy_picture_test PRIVATE ../vendor)
target_link_libraries(av1_legacy_picture_test PRIVATE SteamFrameAV1Backend)
enable_testing()
add_test(NAME legacy_picture_abi_guards COMMAND av1_legacy_picture_test)
add_executable(av1_surface_test surface_test.cpp)
target_link_libraries(av1_surface_test PRIVATE SteamFrameAV1Backend d3d11)
add_executable(av1_legacy_gpu_test legacy_gpu_test.cpp)
target_include_directories(av1_legacy_gpu_test PRIVATE ../vendor)
target_link_libraries(av1_legacy_gpu_test PRIVATE SteamFrameAV1Backend d3d11)
add_executable(av1_trial_control trial_control.cpp)
target_link_libraries(av1_trial_control PRIVATE psapi)
