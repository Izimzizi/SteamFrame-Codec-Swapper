cmake_minimum_required(VERSION 3.20)
project(SteamFrameH264Hook LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SteamFrameH264Hook SHARED legacy_picture.cpp legacy_api_bridge.cpp hook_entry.cpp
 ../vendor/minhook/src/hook.c ../vendor/minhook/src/buffer.c ../vendor/minhook/src/trampoline.c ../vendor/minhook/src/hde/hde64.c)
target_include_directories(SteamFrameH264Hook PRIVATE ../vendor ../vendor/minhook/include)
target_link_libraries(SteamFrameH264Hook PRIVATE d3d11)
target_compile_options(SteamFrameH264Hook PRIVATE /W4 /permissive- /EHsc)
add_executable(picture_abi_test picture_abi_test.cpp)
target_include_directories(picture_abi_test PRIVATE ../vendor)
target_link_libraries(picture_abi_test PRIVATE SteamFrameH264Hook)
enable_testing()
add_test(NAME legacy_picture_abi_guards COMMAND picture_abi_test)
add_executable(bridge_gpu_test bridge_gpu_test.cpp)
target_include_directories(bridge_gpu_test PRIVATE ../vendor)
target_link_libraries(bridge_gpu_test PRIVATE SteamFrameH264Hook d3d11)
add_executable(hook_control hook_control.cpp)
target_link_libraries(hook_control PRIVATE psapi)
