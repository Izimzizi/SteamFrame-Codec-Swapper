cmake_minimum_required(VERSION 3.20)
project(SteamFrameEncoderProbe LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The encoder probe requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(SteamFrameEncoderProbe SHARED encoder_probe.cpp)
target_include_directories(SteamFrameEncoderProbe PRIVATE vendor)
target_link_libraries(SteamFrameEncoderProbe PRIVATE minhook bcrypt)
target_compile_definitions(SteamFrameEncoderProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(SteamFrameEncoderProbe PRIVATE /W4 /permissive- /EHsc)
endif()
enable_testing()
add_executable(qp_map_tests tests/qp_map_tests.cpp)
add_test(NAME qp_map_preservation COMMAND qp_map_tests)
add_executable(probe_guard_tests tests/probe_guard_tests.cpp)
add_test(NAME probe_absent_driver COMMAND probe_guard_tests $<TARGET_FILE:SteamFrameEncoderProbe>)
add_executable(counter_tests tests/counter_tests.cpp)
target_include_directories(counter_tests PRIVATE vendor)
target_link_libraries(counter_tests PRIVATE minhook bcrypt)
target_compile_definitions(counter_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME metadata_only_video_counter COMMAND counter_tests)
add_executable(inspection_tests tests/inspection_tests.cpp)
target_link_libraries(inspection_tests PRIVATE bcrypt)
target_compile_definitions(inspection_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME adaptive_driver_inspection COMMAND inspection_tests)
